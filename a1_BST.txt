#include <iostream>

#include <stack>

#include <queue>



using namespace std;



// Node structure

struct Node {

    int data;

    Node *left, *right;



    Node(int val) {

        data = val;

        left = right = nullptr;

    }

};



class BST {

public:

    Node* root;



    BST() { root = nullptr; }



    // 1. Insert a node (Recursive)

    Node* insert(Node* node, int val) {

        if (node == nullptr) return new Node(val);



        if (val < node->data)

            node->left = insert(node->left, val);

        else if (val > node->data)

            node->right = insert(node->right, val);



        return node;

    }



    // 2. Recursive Traversals

    void inorder(Node* root) {

        if (!root) return;

        inorder(root->left);

        cout << root->data << " ";

        inorder(root->right);

    }



    void preorder(Node* root) {

        if (!root) return;

        cout << root->data << " ";

        preorder(root->left);

        preorder(root->right);

    }



    void postorder(Node* root) {

        if (!root) return;

        postorder(root->left);

        postorder(root->right);

        cout << root->data << " ";

    }



    // 3. Non-Recursive Traversals (Using Stacks)

    void iterativeInorder(Node* root) {

        stack<Node*> s;

        Node* curr = root;

        while (curr != nullptr || !s.empty()) {

            while (curr != nullptr) {

                s.push(curr);

                curr = curr->left;

            }

            curr = s.top();

            s.pop();

            cout << curr->data << " ";

            curr = curr->right;

        }

    }



    void iterativePreorder(Node* root) {

        if (!root) return;

        stack<Node*> s;

        s.push(root);

        while (!s.empty()) {

            Node* curr = s.top();

            s.pop();

            cout << curr->data << " ";

            if (curr->right) s.push(curr->right);

            if (curr->left) s.push(curr->left);

        }

    }



    void iterativePostorder(Node* root) {

        if (!root) return;

        stack<Node*> s1, s2;

        s1.push(root);

        while (!s1.empty()) {

            Node* curr = s1.top();

            s1.pop();

            s2.push(curr);

            if (curr->left) s1.push(curr->left);

            if (curr->right) s1.push(curr->right);

        }

        while (!s2.empty()) {

            cout << s2.top()->data << " ";

            s2.pop();

        }

    }



    // 4. Level-wise Printing (BFS)

    void levelOrder(Node* root) {

        if (!root) return;

        queue<Node*> q;

        q.push(root);

        while (!q.empty()) {

            Node* curr = q.front();

            q.pop();

            cout << curr->data << " ";

            if (curr->left) q.push(curr->left);

            if (curr->right) q.push(curr->right);

        }

    }



    // 5. Delete a Node

    Node* findMin(Node* node) {

        while (node->left != NULL) node = node->left;

        return node;

    }



    Node* deleteNode(Node* root, int key) {

        if (!root) return root;



        if (key < root->data)

            root->left = deleteNode(root->left, key);

        else if (key > root->data)

            root->right = deleteNode(root->right, key);

        else {

            // Case 1 & 2: One child or no child

            if (!root->left) {

                Node* temp = root->right;

                delete root;

                return temp;

            } else if (!root->right) {

                Node* temp = root->left;

                delete root;

                return temp;

            }

            // Case 3: Two children

            Node* temp = findMin(root->right);

            root->data = temp->data;

            root->right = deleteNode(root->right, temp->data);

        }

        return root;

    }

};



int main() {

    BST tree;

    int choice, val;



    do {

        cout << "\n--- BST Operations Menu ---";

        cout << "\n1. Insert Node\n2. Recursive Traversals\n3. Non-Recursive Traversals";

        cout << "\n4. Level-wise Printing\n5. Delete Node\n6. Exit\nEnter Choice: ";

        cin >> choice;



        switch (choice) {

            case 1:

                cout << "Enter value to insert: ";

                cin >> val;

                tree.root = tree.insert(tree.root, val);

                break;

            case 2:

                cout << "\nInorder: "; tree.inorder(tree.root);

                cout << "\nPreorder: "; tree.preorder(tree.root);

                cout << "\nPostorder: "; tree.postorder(tree.root);

                cout << endl;

                break;

            case 3:

                cout << "\nIterative Inorder: "; tree.iterativeInorder(tree.root);

                cout << "\nIterative Preorder: "; tree.iterativePreorder(tree.root);

                cout << "\nIterative Postorder: "; tree.iterativePostorder(tree.root);

                cout << endl;

                break;

            case 4:

                cout << "\nLevel-wise: "; tree.levelOrder(tree.root);

                cout << endl;

                break;

            case 5:

                cout << "Enter value to delete: ";

                cin >> val;

                tree.root = tree.deleteNode(tree.root, val);

                break;

        }

    } while (choice != 6);



    return 0;

}


1. Theory: What is a Binary Search Tree?

A Binary Search Tree (BST) is a node-based binary tree data structure which has the following properties:

The left subtree of a node contains only nodes with keys less than the node’s key.
The right subtree of a node contains only nodes with keys greater than the node’s key.
The left and right subtree must also be a binary search tree.
There must be no duplicate nodes (though some implementations allow them; yours filters them by doing nothing if val == node->data).
Complexity Analysis
Operation	Average Case	Worst Case (Skewed Tree)
Search / Insert / Delete	O(logn)	O(n)
Space	O(n)	O(n)
2. Algorithms and Logic
A. Insertion (Recursive)

To maintain the BST property, the algorithm decides where the new value belongs by comparing it to the current node.

Base Case:
If the current node is NULL, create a new node and return it.
Recursive Step:
If val < current->data, move to the left child.
If val > current->data, move to the right child.
Link the returned nodes properly to reconstruct the tree structure.
B. Depth-First Search (DFS) Traversals

These traversals visit nodes in a specific order using recursion or a stack.

Inorder (L - Root - R):
Visits nodes in ascending order.
Preorder (Root - L - R):
Useful for creating a copy of the tree.
Postorder (L - R - Root):
Useful for deleting the tree or evaluating postfix expressions.
C. Level-Order Traversal (BFS)

Unlike DFS, this visits all nodes at the current depth before moving to the next level.

Algorithm:

Initialize an empty Queue.
Push the root into the queue.
While the queue is not empty:
Pop the front node and print its data.
Push its left child to the queue (if it exists).
Push its right child to the queue (if it exists).
D. Deletion Logic

Deletion is the most complex BST operation because it requires maintaining the tree structure. There are three cases:

Node is a Leaf:
Simply delete it (handled by Case 1 or 2 in your code).
Node has One Child:
Replace the node with its child and delete the original node.
Node has Two Children:
Find the Inorder Successor (smallest value in the right subtree).
Copy the successor’s data into the current node.
Delete the successor node from the right subtree.
